cmake_minimum_required(VERSION 3.22)
project(RingServer VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Supported backends: Linux epoll/uring, macOS kqueue")
endif()
option(RINGSERVER_ENABLE_URING "Build the Linux io_uring backend" OFF)
option(RINGSERVER_ENABLE_KQUEUE_COMPAT "Build Linux libkqueue compatibility (not native kqueue)" OFF)
add_library(ringserver_lib
    src/common/status.cpp src/io/io_backend.cpp
    src/server/connection.cpp src/server/server.cpp)
target_include_directories(ringserver_lib PUBLIC include ../common)
target_compile_options(ringserver_lib PRIVATE -Wall -Wextra -Wpedantic)
if(APPLE)
    target_sources(ringserver_lib PRIVATE src/io/kqueue_backend.cpp)
    target_compile_definitions(ringserver_lib PRIVATE RINGSERVER_PLATFORM_APPLE=1)
else()
    target_sources(ringserver_lib PRIVATE src/io/epoll_backend.cpp)
endif()
if(RINGSERVER_ENABLE_KQUEUE_COMPAT)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "kqueue-compat is Linux-only; Apple builds use native kqueue")
    endif()
    include(../common/kqueue_dependencies.cmake)
    target_sources(ringserver_lib PRIVATE src/io/kqueue_backend.cpp)
    target_link_libraries(ringserver_lib PUBLIC async_libkqueue)
    target_compile_definitions(ringserver_lib PRIVATE RINGSERVER_KQUEUE_COMPAT=1)
endif()
if(RINGSERVER_ENABLE_URING)
    if(APPLE)
        message(FATAL_ERROR "io_uring requires Linux")
    endif()
    include(../common/dependencies.cmake)
    target_sources(ringserver_lib PRIVATE src/io/uring_backend.cpp)
    target_link_libraries(ringserver_lib PUBLIC async_liburing)
    target_compile_definitions(ringserver_lib PRIVATE RINGSERVER_ENABLE_URING=1)
endif()
add_executable(ringserver src/main.cpp)
target_link_libraries(ringserver PRIVATE ringserver_lib)
add_executable(ringserver_test tests/http/http_parser_test.cpp)
target_link_libraries(ringserver_test PRIVATE ringserver_lib)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(ringserver_bench bench/http/http_bench.cpp)
    target_include_directories(ringserver_bench PRIVATE ../common)
endif()
include(CTest)
add_test(NAME http_parser COMMAND ringserver_test)
set_tests_properties(http_parser PROPERTIES TIMEOUT 10)
