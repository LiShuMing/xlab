cmake_minimum_required(VERSION 3.22)
project(async_uring LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(../common/dependencies.cmake)
add_library(io_uring INTERFACE)
target_include_directories(io_uring INTERFACE include ../common)
target_link_libraries(io_uring INTERFACE async_liburing)
include(CTest)
foreach(name basic_test delay_and_print ping_pong)
    add_executable(test_${name} tests/${name}.cpp)
    target_link_libraries(test_${name} PRIVATE io_uring)
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 15 SKIP_RETURN_CODE 77)
endforeach()
add_executable(uring_bench bench/uring_bench.cpp)
target_link_libraries(uring_bench PRIVATE io_uring)
add_executable(uring_probe tests/probe.cpp)
target_link_libraries(uring_probe PRIVATE io_uring)
